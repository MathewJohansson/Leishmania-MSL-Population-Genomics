python3 -c "

# Run samtools bedcov for all five samples
module load SAMtools
for s in SRR20748483 SRR20748320 SRR10478771 SRR10478777 ERR3550134; do
    samtools bedcov MSL_locus_100nt_windows.bed bams/${s}.sort_no_dup.bam > coverage_${s}.txt
    echo "$s done"
done
